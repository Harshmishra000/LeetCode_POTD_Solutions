class Solution {
    public int sumSubarrayMins(int[] arr) {
       int n = arr.length;
       long sum = 0;
       for(int i = 0; i< n; i++){
            int min = arr[i];
            for(int j = i; j< n; j++){
                min = Math.min(min, arr[j]);
                sum += min; 
            }
       }
       return (int)(sum % 1000000007);
    }
}
