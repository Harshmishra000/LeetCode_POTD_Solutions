class Solution {
    public long subArrayRanges(int[] nums) {
      int n = nums.length;

      long sum = 0;
      for(int i = 0; i< n; i++){
        int small = nums[i], large = nums[i];
        for(int j = i; j< n; j++){
            small = Math.min(small, nums[j]);
            large = Math.max(large, nums[j]);
            sum +=  large - small;
        }
      }
      return sum;  
    }
}
